var firstID = "sha256:" + strings.Repeat("a", 64)
var secondID = "sha256:" + strings.Repeat("b", 64)

type fakeEngine struct {
	mu                             sync.Mutex
	local                          map[string]string
	pullID                         string
	pulls                          int
	inspectErr, pullErr, retainErr error
	started, unblock               chan struct{}
}

func (f *fakeEngine) Inspect(_ context.Context, ref string) (string, error) {
	f.mu.Lock()
	defer f.mu.Unlock()
	if f.inspectErr != nil {
		return "", f.inspectErr
	}
	if id := f.local[ref]; id != "" {
		return id, nil
	}
	return "", ErrMissing
}
func (f *fakeEngine) Pull(ctx context.Context, ref string) (string, error) {
	f.mu.Lock()
	f.pulls++
	id, err := f.pullID, f.pullErr
	started, unblock := f.started, f.unblock
	f.mu.Unlock()
	if started != nil {
		select {
		case started <- struct{}{}:
		default:
		}
	}
	if unblock != nil {
		select {
		case <-unblock:
		case <-ctx.Done():
			return "", ctx.Err()
		}
	}
	f.mu.Lock()
	defer f.mu.Unlock()
	if err == nil {
		f.local[ref] = id
		f.local[id] = id
	}
	return id, err
}
func (f *fakeEngine) Retain(_ context.Context, id, tag string) error {
	f.mu.Lock()
	defer f.mu.Unlock()
	if f.retainErr != nil {
		return f.retainErr
	}
	f.local[tag] = id
	return nil
}
func newTest(t *testing.T, cached bool) (*Manager, *fakeEngine) {
	t.Helper()
	f := &fakeEngine{local: map[string]string{}, pullID: firstID}
	if cached {
		f.local["example/game:stable"] = firstID
		f.local[firstID] = firstID
	}
	m, err := New(f, filepath.Join(t.TempDir(), "state.json"))
	if err != nil {
		t.Fatal(err)
	}
	return m, f
}
func resolve(t *testing.T, m *Manager) string {
	t.Helper()
	id, err := m.Resolve(context.Background(), "example/game:stable")
	if err != nil {
		t.Fatal(err)
	}
	return id
}
func waitIdle(t *testing.T, m *Manager) {
	t.Helper()
	deadline := time.Now().Add(3 * time.Second)
	for time.Now().Before(deadline) {
		rows, err := m.Snapshot()
		if err != nil {
			t.Fatal(err)
		}
		if len(rows) > 0 && !rows[0].Busy {
			return
		}
		time.Sleep(time.Millisecond)
	}
	t.Fatal("manager remained busy")
}
func TestCachedStartHasNoPullAndSurvivesRestart(t *testing.T) {
	m, f := newTest(t, true)
	if resolve(t, m) != firstID || f.pulls != 0 {
		t.Fatal("cached image pulled")
	}
	again, err := New(f, m.path)
	if err != nil {
		t.Fatal(err)
	}
	f.local["example/game:stable"] = secondID
	if resolve(t, again) != firstID || f.pulls != 0 {
		t.Fatal("mutable tag changed approved image")
	}
}
func TestColdConcurrentStartsDeduplicate(t *testing.T) {
	m, f := newTest(t, false)
	f.unblock = make(chan struct{})
	f.started = make(chan struct{}, 1)
	var wg sync.WaitGroup
	errs := make(chan error, 12)
	for i := 0; i < 12; i++ {
		wg.Add(1)
		go func() {
			defer wg.Done()
			id, err := m.Resolve(context.Background(), "example/game:stable")
			if err == nil && id != firstID {
				err = errors.New("wrong identity")
			}
			errs <- err
		}()
	}
	<-f.started
	close(f.unblock)
	wg.Wait()
	close(errs)
	for err := range errs {
		if err != nil {
			t.Fatal(err)
		}
	}
	if f.pulls != 1 {
		t.Fatalf("pulls=%d", f.pulls)
	}
}

func TestStagingApprovalRollbackAndStaleSelection(t *testing.T) {
	m, f := newTest(t, true)
	resolve(t, m)
	f.pullID = secondID
	if err := m.StartCheck("example/game:stable"); err != nil {
		t.Fatal(err)
	}
	waitIdle(t, m)
	rows, _ := m.Snapshot()
	if rows[0].CandidateID != secondID || rows[0].LastCheckedAt == nil || resolve(t, m) != firstID {
		t.Fatal("candidate activated before approval")
	}
	if m.Change(context.Background(), "example/game:stable", "approve", firstID) == nil {
		t.Fatal("stale approval accepted")
	}
	if err := m.Change(context.Background(), "example/game:stable", "approve", secondID); err != nil {
		t.Fatal(err)
	}
	if resolve(t, m) != secondID {
		t.Fatal("approval not used")
	}
	if err := m.Change(context.Background(), "example/game:stable", "rollback", firstID); err != nil {
		t.Fatal(err)
	}
	if resolve(t, m) != firstID {
		t.Fatal("rollback not used")
	}
}
func TestCachedStartDoesNotWaitForUpdateAndCheckIsDeduplicated(t *testing.T) {
	m, f := newTest(t, true)

	f.pullID = secondID
	f.started = make(chan struct{}, 1)
	f.unblock = make(chan struct{})
	if err := m.StartCheck("example/game:stable"); err != nil {
		t.Fatal(err)
	}
	<-f.started
	if !errors.Is(m.StartCheck("example/game:stable"), ErrBusy) {
		t.Fatal("duplicate check")
	}
	ctx, cancel := context.WithTimeout(context.Background(), 100*time.Millisecond)
	defer cancel()
	id, err := m.Resolve(ctx, "example/game:stable")
	if err != nil || id != firstID {
		t.Fatal("cached start waited for pull", err)
	}
	close(f.unblock)
	waitIdle(t, m)
}
func TestCheckFailureRetainsApprovalAndPersistsBackoff(t *testing.T) {
	m, f := newTest(t, true)
	resolve(t, m)
	f.pullErr = errors.New("private registry secret must not leak")
	stamp := time.Date(2026, 11, 5, 0, 0, 0, 0, time.UTC)
	m.now = func() time.Time { return stamp }
	if err := m.StartCheck("example/game:stable"); err != nil {
		t.Fatal(err)
	}
	waitIdle(t, m)
	rows, _ := m.Snapshot()
	s := rows[0]
	if s.ApprovedID != firstID || s.LastCheckedAt != nil || s.NextCheckAt.Sub(stamp) != time.Hour || s.Error == "" || strings.Contains(s.Error, "secret") {
		t.Fatal(s)
	}
	again, err := New(f, m.path)
	if err != nil {
		t.Fatal(err)
	}
	if resolve(t, again) != firstID {
		t.Fatal("failed check broke cached boot")
	}
}

func TestApprovalAndChecksCannotOverwriteEachOther(t *testing.T) {
	m, f := newTest(t, true)
	resolve(t, m)
	f.pullID = secondID
	if err := m.StartCheck("example/game:stable"); err != nil {
		t.Fatal(err)
	}
	waitIdle(t, m)
	release, err := m.claim(context.Background(), "example/game:stable", false)
	if err != nil {
		t.Fatal(err)
	}
	if !errors.Is(m.StartCheck("example/game:stable"), ErrBusy) {
		t.Fatal("check entered approval critical section")
	}
	if !errors.Is(m.Schedule(context.Background(), "example/game:stable", false), ErrBusy) {
		t.Fatal("schedule entered approval critical section")
	}
	s, _ := m.state("example/game:stable")
	s.PreviousID, s.ApprovedID, s.CandidateID = s.ApprovedID, s.CandidateID, ""
	if err = m.commit(s); err != nil {
		t.Fatal(err)
	}
	release()
	if err = m.StartCheck("example/game:stable"); err != nil {
		t.Fatal(err)
	}
	waitIdle(t, m)
	rows, _ := m.Snapshot()
	if rows[0].ApprovedID != secondID || rows[0].PreviousID != firstID {
		t.Fatal("check reverted completed approval")
	}
}

type waitSignalContext struct {
	context.Context
	ready chan struct{}
	once  sync.Once
}

func (c *waitSignalContext) Done() <-chan struct{} {
	c.once.Do(func() { close(c.ready) })
	return c.Context.Done()
}
func TestConcurrentColdFailureSharedWithWaiters(t *testing.T) {
	m, f := newTest(t, false)
	f.pullErr = errors.New("registry unavailable")
	f.started = make(chan struct{}, 1)
	f.unblock = make(chan struct{})
	leader := make(chan error, 1)
	go func() { _, err := m.Resolve(context.Background(), "example/game:stable"); leader <- err }()
	<-f.started
	ctx := &waitSignalContext{Context: context.Background(), ready: make(chan struct{})}
	follower := make(chan error, 1)
	go func() {
		release, err := m.claim(ctx, "example/game:stable", true)
		if release != nil {
			release()
		}
		follower <- err
	}()
	<-ctx.ready
	close(f.unblock)
	if <-leader == nil || <-follower == nil {
		t.Fatal("failed in-flight download was not shared")
	}
	if f.pulls != 1 {
		t.Fatal("failed download was retried by waiter")
	}
}
