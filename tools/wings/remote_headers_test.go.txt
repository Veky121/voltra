func TestVoltraRemoteHeaders(t *testing.T) {
	u, _ := url.Parse("https://cdn25.giants-software.com/mod.zip")
	for _, headers := range []map[string]string{
		{"Authorization": "secret"}, {"Cookie": "secret"}, {"Host": "localhost"}, {"Range": "bytes=0-1"},
		{"Referer": "https://example.com/?key=secret"}, {"Referer": "http://example.com/"},
		{"Accept": "text/plain\r\nInjected: bad"}, {"Referer": "https://example.com/", "referer": "https://other.test/"},
	} {
		if ValidateRemoteHeaders(u, headers) == nil {
			t.Fatalf("accepted %#v", headers)
		}
	}
	headers := map[string]string{"Referer": "https://www.farming-simulator.com/"}
	for _, raw := range []string{"http://example.com/mod.zip", "https://user@example.com/mod.zip", "https://example.com:444/mod.zip"} {
		target, _ := url.Parse(raw)
		if ValidateRemoteHeaders(target, headers) == nil {
			t.Fatal(raw)
		}
	}
}

func TestVoltraRemoteHeadersRequestAndRedirect(t *testing.T) {
	var targetCalls int
	target := httptest.NewTLSServer(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) { targetCalls++; w.WriteHeader(200) }))
	defer target.Close()
	source := httptest.NewTLSServer(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		if r.Header.Get("Referer") != "https://www.farming-simulator.com/" {
			t.Error("Referer missing")
		}
		http.Redirect(w, r, target.URL, http.StatusFound)
	}))
	defer source.Close()
	// Replace only the transport for TLS fixture connectivity. Keep the production
	// redirect policy, which must prevent all cross-host/header forwarding.
	c := *client
	c.Transport = source.Client().Transport
	req, _ := http.NewRequest(http.MethodGet, "https://cdn25.giants-software.com/mod.zip", nil)
	if err := applyRemoteHeaders(req, map[string]string{"Referer": "https://www.farming-simulator.com/"}); err != nil {
		t.Fatal(err)
	}
	fixture, _ := url.Parse(source.URL)
	req.URL = fixture
	res, err := c.Do(req)
	if err != nil {
		t.Fatal(err)
	}
	io.Copy(io.Discard, res.Body)
	res.Body.Close()
	if res.StatusCode != 302 || targetCalls != 0 {
		t.Fatalf("redirect followed: %d calls=%d", res.StatusCode, targetCalls)
	}
}
