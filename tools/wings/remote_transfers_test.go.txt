package downloader

import (
 "io"
 "net/http"
 "net/url"
 "strings"
 "testing"
 "time"

 "github.com/pterodactyl/wings/server"
)

type transferTransport func(*http.Request)(*http.Response,error)
func(f transferTransport) RoundTrip(r *http.Request)(*http.Response,error){return f(r)}

func TestVoltraRemoteFailureRetainedAndEmitted(t *testing.T){
 s,err:=server.New(nil);if err!=nil {t.Fatal(err)}
 defer s.CleanupForDestroy()
 events:=make(chan []byte,16);s.Events().On(events);defer s.Events().Off(events)
 saved:=client;defer func(){client=saved}()
 client=&http.Client{Transport:transferTransport(func(r *http.Request)(*http.Response,error){return &http.Response{StatusCode:403,Body:io.NopCloser(strings.NewReader("private source body")),Header:make(http.Header)},nil})}
 u,_:=url.Parse("https://example.com/mod.zip?secret=never-expose")
 dl:=New(s,DownloadRequest{Directory:"/mods",URL:u})
 if dl.Execute()==nil {t.Fatal("403 reported success")}
 if ByID(dl.Identifier)!=nil {t.Fatal("failed download still occupies a concurrency slot")}
 history:=TransferHistory(s.ID());var found *TransferRecord
 for i:=range history {if history[i].Identifier==dl.Identifier {found=&history[i]}}
 if found==nil||found.State!="failed"||found.Error!="Download site returned 403 Forbidden"||found.Directory!="/mods" {t.Fatalf("missing terminal outcome: %#v",found)}
 // The terminal event must be delivered even when no bytes were downloaded.
 timer:=time.NewTimer(time.Second);defer timer.Stop()
 for {select {case data:=<-events:
  if strings.Contains(string(data),"never-expose")||strings.Contains(string(data),"private source body"){t.Fatal("source secret leaked")}
  if strings.Contains(string(data),"failed"){return}
 case <-timer.C:t.Fatal("missing failure event")}}
}
