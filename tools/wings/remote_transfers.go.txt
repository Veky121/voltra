package downloader

import (
	"context"
	"encoding/json"
	"errors"
	"io"
	"net"
	"net/http"
	"os"
	"path"
	"sort"
	"sync"
	"time"
)

type TransferRecord struct {
	Identifier    string  `json:"identifier"`
	Directory     string  `json:"directory"`
	Name          string  `json:"name"`
	State         string  `json:"state"`
	Progress      float64 `json:"progress"`
	ReceivedBytes int64   `json:"received_bytes"`
	TotalBytes    int64   `json:"total_bytes"`
	SizeKnown     bool    `json:"size_known"`
	Error         string  `json:"error,omitempty"`
	Updated       int64   `json:"updated"`
}
type transferHistory struct {
	sync.Mutex
	records map[string]map[string]TransferRecord
}

var transfers = transferHistory{records: make(map[string]map[string]TransferRecord)}

func TransferHistory(server string) []TransferRecord {
	transfers.Lock()
	defer transfers.Unlock()
	result := []TransferRecord{}
	for id, r := range transfers.records[server] {
		if r.State != "downloading" && r.State != "queued" && time.Now().UnixMilli()-r.Updated > 15*60*1000 {
			delete(transfers.records[server], id)
			continue
		}
		result = append(result, r)
	}
	sort.Slice(result, func(i, j int) bool { return result[i].Updated < result[j].Updated })
	return result
}

func (dl *Download) recordTransfer(state string, progress float64, err error) {
	name := dl.req.FileName
	if name == "" {
		name = path.Base(dl.req.URL.Path)
	}
	if dl.path != "" {
		name = dl.path
	}
	r := TransferRecord{Identifier: dl.Identifier, Directory: dl.req.Directory, Name: name, State: state, Progress: progress, Updated: time.Now().UnixMilli()}
	dl.mu.RLock()
	r.ReceivedBytes, r.TotalBytes = dl.receivedBytes, dl.totalBytes
	dl.mu.RUnlock()
	r.SizeKnown = r.TotalBytes >= 0 && state != "queued"
	if !r.SizeKnown {
		r.TotalBytes = 0
	}
	if err != nil {
		var source *RemoteStatusError
		if errors.As(err, &source) {
			r.Error = source.Error()
		} else {
			r.Error = transferFailureMessage(err)
		}
	}
	sid := dl.server.ID()
	transfers.Lock()
	if transfers.records[sid] == nil {
		transfers.records[sid] = make(map[string]TransferRecord)
	}
	old := transfers.records[sid][dl.Identifier]
	if state == "downloading" && old.State == state && r.Updated-old.Updated < 500 {
		transfers.Unlock()
		return
	}
	transfers.records[sid][dl.Identifier] = r
	// Bound retained outcomes without evicting active transfers.
	if len(transfers.records[sid]) > 50 {
		oldest := ""
		var stamp int64
		for id, entry := range transfers.records[sid] {
			if entry.State == "complete" || entry.State == "failed" {
				if oldest == "" || entry.Updated < stamp {
					oldest = id
					stamp = entry.Updated
				}
			}
		}
		if oldest != "" {
			delete(transfers.records[sid], oldest)
		}
	}
	transfers.Unlock()
	data, _ := json.Marshal(r)
	dl.server.Events().Publish("download progress", string(data))
}

var errRemoteSizeUnknown = errors.New("download source did not provide a file size")
var errRemoteSizeMismatch = errors.New("download source provided an inconsistent file size")

// A bounded HEAD fallback uses the same SSRF and redirect policy as the actual
// download. It cannot turn an untrusted URL into a request from the panel host.
func (dl *Download) lookupSize(ctx context.Context) int64 {
	ctx, cancel := context.WithTimeout(ctx, 10*time.Second)
	defer cancel()
	req, err := http.NewRequestWithContext(ctx, http.MethodHead, dl.req.URL.String(), nil)
	if err != nil {
		return -1
	}
	req.Header.Set("User-Agent", "Pterodactyl Panel (https://pterodactyl.io)")
	if applyRemoteHeaders(req, dl.req.RemoteHeaders) != nil {
		return -1
	}
	res, err := client.Do(req)
	if err != nil {
		return -1
	}
	defer res.Body.Close()
	if res.StatusCode != http.StatusOK {
		return -1
	}
	return res.ContentLength
}

func transferFailureMessage(err error) string {
	var network net.Error
	switch {
	case errors.Is(err, errRemoteSizeUnknown):
		return "The download site did not provide a file size. This node needs the size to enforce your disk limit. Use a direct download link or download the file locally and upload it."
	case errors.Is(err, errRemoteSizeMismatch):
		return "The download site reported a size that did not match the file. A partial file may remain. Download the file locally and upload it."
	case errors.Is(err, io.ErrUnexpectedEOF):
		return "The download ended before the whole file arrived. A partial file may remain. Retry the download."
	case errors.Is(err, context.Canceled):
		return "The download was canceled. A partial file may remain. Start it again if needed."
	case errors.Is(err, context.DeadlineExceeded), errors.As(err, &network) && network.Timeout():
		return "The download connection timed out. Check the source is available and retry."
	case errors.Is(err, os.ErrPermission):
		return "The node could not write the downloaded file. Ask an administrator to check folder permissions."
	default:
		return "The node could not finish the download. A partial file may remain. Check free disk space and the node logs before retrying."
	}
}
